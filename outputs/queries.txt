=========================================================

{
  "year": 2018,
  "geography": { "geoid": "...", "type": "...", "name": "..." },
  "population": { "total": ..., "condition_weighted": ... },

    "condition_score": ..., "rpl_themes": ..., "quintile": ...,
    "f_total": ..., "weighted_avg_quintile": ..., "not_socially_vulnerable": ...

  "shape": { "area": ..., "length": ... }
}

Q4. What is the average social vulnerability score (rpl_themes) per quintile?
    (Lower rpl_themes = more vulnerable. Grouped and sorted by quintile.)
=========================================================

db.svi.aggregate([
  { $group: {
      _id: "$vulnerability.quintile",
      avg_rpl_themes: { $avg: "$vulnerability.rpl_themes" },
      tract_count:    { $sum: 1 }
  }},
  { $sort: { _id: 1 } }
])

-- OUTPUT --
[
  { _id: 1, avg_rpl_themes: 0.8775525, tract_count: 80 },
  { _id: 2, avg_rpl_themes: 0.6207125, tract_count: 80 },
  { _id: 3, avg_rpl_themes: 0.36600379746835443, tract_count: 79 },

  { _id: 5, avg_rpl_themes: 0.04791645569620253, tract_count: 79 }

Q9. Which quintile has the most census tracts?
    (Useful for understanding the distribution of vulnerability across King County.)
=========================================================

db.svi.aggregate([
  { $group: { _id: "$vulnerability.quintile", count: { $sum: 1 } } },
  { $sort: { count: -1 } }
])

  { _id: 2, count: 80 },

Q10. What is the average population size of census tracts in each quintile,
     sorted from most to least vulnerable?
     (Are vulnerable tracts smaller or larger than average?)
=========================================================

db.svi.aggregate([
  { $group: {
      _id: "$vulnerability.quintile",
      avg_population: { $avg: "$population.total" },
